This section builds on the material in \cref{sec:c3s2}, \cref{ch:4} and the prior work of \citet{ch3-13}; see also \citep{ch3-3,ch3-9}.

\begin{equation}\label{eq:c3s3-a}
  \frac{\partial u}{\partial t} = \nabla\cdot\bigl(D(\mathbf{x})\nabla u\bigr) + f(u),\qquad u(\mathbf{x},0)=u_0(\mathbf{x}).
\end{equation}
Discretising in space with $N$ degrees of freedom yields the semi-discrete system
\begin{align}\label{eq:c3s3-b}
  \mathbf{M}\dot{\mathbf{u}} &= -\mathbf{K}\mathbf{u} + \mathbf{F}(\mathbf{u}),\\
  \mathbf{u}(0) &= \mathbf{u}_0 \in \mathbb{R}^{N}.
\end{align}

Equations~\eqref{eq:c3s3-a} and~\eqref{eq:c3s3-b} are used throughout the remainder of this section.

\subsection{Implementation notes}\label{sec:c3s3-i}
The core routine is shown in \cref{lst:c3s3}; its complexity is $\mathcal{O}(N\log N)$ per iteration~\cite{ch3-14}.

\begin{lstlisting}[language=c,caption={Reference implementation for c3s3.},label={lst:c3s3}]
#include <stdio.h>

/* Compute a dot product of two vectors. */
double dot(const double *a, const double *b, int n) {
    double s = 0.0;
    for (int i = 0; i < n; ++i)
        s += a[i] * b[i];
    return s;
}
\end{lstlisting}

\begin{theorem}\label{thm:c3s3}
Let $A\in\mathbb{R}^{n\times n}$ be symmetric positive definite with condition number $\kappa$. Then the iteration converges and
$\lVert e_k\rVert_A \le 2\bigl(\tfrac{\sqrt\kappa-1}{\sqrt\kappa+1}\bigr)^k\lVert e_0\rVert_A$.
\end{theorem}
\begin{enumerate}[label=(\roman*)]
\item the bound in \cref{thm:c3s3} is sharp for some initial errors;
\item the constant $2$ cannot be improved in general;
\item preconditioning reduces the effective $\kappa$.
\end{enumerate}
